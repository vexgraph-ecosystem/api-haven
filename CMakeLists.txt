cmake_minimum_required(VERSION 3.24)
# Standalone CLion metadata only. Actual builds belong to b; no fetch or link.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported OS floor")
endif()
project(api_haven_ide LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VEXSPOKE_SOURCE_DIR "" CACHE PATH "Local Vexspoke src; no downloads or fake declarations")
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "src/*.c" "src/*.h")
list(FILTER sources EXCLUDE REGEX "/(attic|vendor|third_party|_old)/")
add_library(api_haven_ide OBJECT EXCLUDE_FROM_ALL ${sources})
target_include_directories(api_haven_ide PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(VEXSPOKE_SOURCE_DIR)
    target_include_directories(api_haven_ide PRIVATE "${VEXSPOKE_SOURCE_DIR}")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(api_haven_ide PRIVATE -std=gnu23 -Wall -Wextra -Werror)
    if(APPLE)
        target_compile_options(api_haven_ide PRIVATE -mcpu=apple-m1)
    endif()
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    execute_process(COMMAND "${CMAKE_C_COMPILER}" -print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(EXISTS "${resource}/include/stddef.h")
        target_compile_options(api_haven_ide PRIVATE -isystem "${resource}/include")
    endif()
endif()
message(STATUS "IDE metadata only: excluded object target; supply local dependency paths. Use b for builds.")
